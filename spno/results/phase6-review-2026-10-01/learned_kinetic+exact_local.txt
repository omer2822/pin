from spno.solvers.split_step import SplitStepNLSOperator, SubsteppedReference
from spno.artifacts import atomic_json


class ComponentSwap(LearnedSplitStep):
    def __init__(self, learned, exact_kinetic=False, exact_local=False):
        super().__init__(
            learned.domain, kinetic_mode=None,
            trained_dt=learned.trained_dt,
        )
        self.learned = learned
        self.exact = ExactSplitStep(learned.domain)
        self.exact_kinetic = exact_kinetic
        self.exact_local = exact_local

    def _kinetic_half(self, field, parameters, dt):
        operator = self.exact if self.exact_kinetic else self.learned
        return operator._kinetic_half(field, parameters, dt)

    def local_phase(self, field, potential, alpha, beta):
        operator = self.exact if self.exact_local else self.learned
        return operator.local_phase(field, potential, alpha, beta)


@torch.inference_mode()
def measure_components(model, config, inputs, steps=200, stride=25):
    assert steps > 0 and stride > 0 and steps % stride == 0

    model = widen_to_double(model, device="cpu").eval()
    both_exact = ComponentSwap(model, True, True)

    # Check that the wrapper preserves the intended compositions.
    torch.testing.assert_close(
        ComponentSwap(model)(*inputs), model(*inputs),
        atol=1e-12, rtol=1e-12,
    )
    torch.testing.assert_close(
        both_exact(*inputs),
        SplitStepNLSOperator(config.domain)(*inputs),
        atol=1e-12, rtol=1e-12,
    )

    variants = {
        "reference": SubsteppedReference(config.domain, config.substeps),
        "C1": model,
        "exact kinetic + learned local": ComponentSwap(model, True, False),
        "learned kinetic + exact local": ComponentSwap(model, False, True),
        "both exact": both_exact,
    }

    results = {}
    for name, operator in variants.items():
        print(f"Running: {name}", flush=True)
        result = cascade_series(
            operator.eval(), config.domain, *inputs,
            steps=steps, stride=stride,
            cutoff=float(config.initial_bandwidth),
        )
        assert result["steps"][-1] == steps, f"{name} stopped early"
        results[name] = result

    return results

from dataclasses import asdict
from datetime import datetime, timezone
import hashlib
import zipfile


def load_c1_seed(seed):
    filename = f"C1-seed{seed}.pt"
    relative = (
        Path("checkpoints/phase6")
        / f"{config_hash(config)}-K0"
        / filename
    )
    path = PROJECT_ROOT / "results/phase6-standalone-artifacts" / relative
    cached = Path("/content/c1-diagnostics") / filename

    if not path.is_file():
        path = cached

    if not path.is_file():
        archive_names = [
            "phase6-eval-only-bd4e108527-K0.zip",
            "phase6-standalone-artifacts-bd4e108527-K0.zip",
        ]
        archive_path = next(
            (
                p
                for name in archive_names
                for p in Path("/content/drive/MyDrive").rglob(name)
            ),
            None,
        )
        if archive_path is None:
            raise FileNotFoundError(f"No checkpoint or archive found for seed {seed}")

        with zipfile.ZipFile(archive_path) as archive:
            members = [
                name for name in archive.namelist()
                if name.endswith("/" + relative.as_posix())
            ]
            assert len(members) == 1, f"Checkpoint match is ambiguous: {members}"
            path.parent.mkdir(parents=True, exist_ok=True)
            path.write_bytes(archive.read(members[0]))

    saved = load_checkpoint_payload(path)
    meta = saved.metadata
    assert meta.model_name == "C1" and meta.seed == seed

    arch = meta.architecture
    model = DensityPhaseSplitStep(
        config.domain,
        kinetic_mode=arch.get("kinetic_mode", "K0"),
        local_mode=arch.get("local_mode") or "L0",
        width=int(arch.get("width", 32)),
        trained_dt=meta.trained_dt,
    )
    model = restore_checkpoint(
        model, saved,
        expected_data_hash=config_hash(config),
        allow_budget_bound=True,
    ).eval()

    return model, saved, path
